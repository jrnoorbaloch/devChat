A useful bot in 20 lines

Plug in the built-in CommandRouter and AutoReply plugins. The same code runs on every messenger you picked in npm run setup.

Read the plugin guide →

import { startDevChat, CommandRouter, AutoReply } from 'devchat'

await startDevChat({
  setup: (bot, messenger) => {
    bot.use(
      CommandRouter({
        commands: {
          ping: () => 'pong',
          echo: ({ rawArgs }) => rawArgs,
        },
        rateLimit: { limit: 10, intervalMs: 60_000 },
      }),
      AutoReply({
        rules: [{ match: 'hi', reply: 'Hello! Type /help' }],
      }),
    )
  },
})
// npm start → scan the QR code on WhatsApp / WeChat
